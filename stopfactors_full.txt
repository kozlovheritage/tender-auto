294: def _is_reliable_brand(brand_hint: str) -> bool:
295:     """
296:     Возвращает True если brand_hint — конкретное название производителя,
297:     для которого имеет смысл запускать поиск поставщиков на мировом рынке.
298: 
299:     Правила:
300:     - None / пустая строка / < 3 символов → False
301:     - Содержит «информация отсутствует», «определено автоматически»,
302:       «требует проверки», «заполняется вручную» → False (ЛЛМ не уверен)
303:     - Строка длиннее 80 символов → скорее описание, не бренд → False
304:     - Нет ни одной латинской буквы → кириллическое нарицательное или
305:       российский производитель; мировой поиск бессмысленен → False
306:     """
307:     if not brand_hint:
308:         return False
309:     b = brand_hint.strip()
310:     if len(b) < 3:
311:         return False
312:     b_lower = b.lower()
313:     for bad in ('информация отсутствует', 'определено автоматически',
314:                 'требует проверки', 'заполняется вручную'):
315:         if bad in b_lower:
316:             return False
317:     if len(b) > 80:
318:         return False
319:     # Требуем хотя бы одну латинскую букву — реальный бренд для глобального
320:     # рынка почти всегда имеет латинское написание
321:     if not re.search(r'[A-Za-z]', b):
322:         return False
323:     return True
